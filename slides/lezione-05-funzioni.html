🔁 Return

Restituire un valore

⚖️ Differenza importante

print() vs return

print() — mostra, non restituisce


def doppio(n):
    print(n * 2)

x = doppio(5)
print(x)      # None

return — restituisce il valore


def doppio(n):
    return n * 2

x = doppio(5)
print(x)      # 10

    if n > 0:
        return "positivo"
    elif n < 0:
        return "negativo"
    else:
        return "zero"

def massimo(a, b):

💻 Analizzatore di testo

def tokenizza(testo): return testo.split() def conta_parole(parole): return len(parole) def parola_piu_lunga(parole): lunga = parole[0] for p in parole[1:]: if len(p) > len(lunga): lunga = p return lunga def lunghezza_media(parole): totale = 0 for p in parole: totale = totale + len(p) return totale / len(parole) 

from js import prompt

testo = prompt("Inserisci una frase:")
parole = tokenizza(testo)

print("Numero parole:", conta_parole(parole))

📝 parola_piu_lunga


def tokenizza(testo):
    return testo.split()

def parola_piu_lunga(parole):
    lunga = parole[0]
    for p in parole:
        if len(p) > len(lunga):
            lunga = p
    return lunga

parole = tokenizza("ciao mondo come stai")
print(parola_piu_lunga(parole))
Soluzione

📝 lunghezza_media


def tokenizza(testo):
    return testo.split()

def lunghezza_media(parole):
    totale = 0
    for p in parole:
        totale = totale + len(p)
    return totale / len(parole)

parole = tokenizza("ciao mondo come stai")
print(lunghezza_media(parole))
